model{
  #Likelihood  
  lambda[1] <- 0
  
  for (i in 2:n){
    N[i] ~ dlnorm(log(max(lambda[i]*N[i-1] - removals[i], 0.001)), tauState)
  
    y[i] ~ dlnorm(log(N[i]) - 1/(2*tauObs[i]), tauObs[i])
  
    log(lambda[i]) <- beta0 + beta1 * rain[i]

  }
  
  #Priors
  
  N[1] ~ dunif(0, 2)
  
  y[1] ~ dlnorm(log(N[1]) - 1/(2*tauObs[1]), tauObs[1])
  
  beta0 ~ dnorm(0, 0.01)
  beta1 ~ dnorm(0, 0.01)
  
  tauState ~ dgamma(0.1, 0.1)
}
