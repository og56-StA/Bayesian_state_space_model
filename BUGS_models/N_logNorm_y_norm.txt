model{
  #Likelihood  
  lambda[1] <- 0
  
  for (i in 2:n){
    N[i] ~ dlnorm(log(max(lambda[i]*N[i-1] - removals[i], 0.001)), tauState)
  
    log(lambda[i]) <- beta0 + beta1 * rain[i]

  }

  for (i in obsIdx){
	y[i] ~ dnorm(N[i], tauObs[i])
  }
  
  #Priors
  
  N[1] ~ dunif(0, 2)
    
  beta0 ~ dnorm(0, 0.01)
  beta1 ~ dnorm(0, 0.01)
  
  tauState ~ dgamma(0.1, 0.1) T(1,) #Changed as chains were getting stuck at 0 precision RC
  
  #Derived quantities added for predictions - RC
  lambdaAvg  <- exp(beta0)         #growth multiplier at average rainfall
  pBeta1Pos  <- step(beta1)        #1 if beta1 > 0; its mean = Pr(beta1 > 0)
}
